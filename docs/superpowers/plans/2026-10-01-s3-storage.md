# S3 兼容存储支持 — 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: 用 superpowers:subagent-driven-development 按本计划执行。步骤用 `- [ ]` 跟踪。规范（冻结接口）在 `docs/superpowers/specs/2026-10-01-s3-storage-design.md`，本计划与它冲突时以 spec 为准并上报 lead。

**Goal:** 内核新增可切换存储驱动：默认 `local` 行为逐字节不变；`storage.driver=s3` 时成品落 S3 兼容对象存储（零新 composer 依赖、手写 SigV4 客户端），读取走 302 预签名，覆盖 AWS/各大云/自托管 40+ 家 S3 兼容厂商。

**Architecture:** 新增 `src/Storage/`（StorageInterface + LocalStorage + S3Storage + S3Client + SignatureV4 + Storage 门面）；三处接线：`PartialResource::rename()` → `publish()`、`Resource::exists()/delete()` → driver、`ResourceController` 读取 → `url()` 非空走 302。配置在 `config/app.php`、`config/aetherupload.php`、Symfony `Configuration` 树三处同步。

**Tech Stack:** PHP 8.0–8.4（禁 enum/readonly/never）· PHPUnit 9.6+10.5 双版本 · ext-curl（S3 模式软依赖）· MinIO（docker）端到端。

```php
class SignatureV4Test extends TestCase
{
    const ACCESS = 'AKIDEXAMPLE';
    const SECRET = 'wJalrXUtnFEMI/K7MDENG+bPxRfiCYEXAMPLEKEY';
    const AMZ_DATE = '20150830T123600Z';
    const REGION = 'us-east-1';
    const SERVICE = 'service';
    const EMPTY_HASH = 'e3b0c44298fc1c149afbf4c8996fb92427ae41e4649b934ca495991b7852b855';

    public function testGetVanillaVector(): void
    {
        $headers = ['Host' => 'example.amazonaws.com', 'X-Amz-Date' => self::AMZ_DATE];

        $creq = SignatureV4::canonicalRequest('GET', '/', '', $headers, self::EMPTY_HASH);

        $this->assertSame(
            "GET\n/\n\nhost:example.amazonaws.com\nx-amz-date:" . self::AMZ_DATE

        );
        $this->assertSame(
            'AWS4-HMAC-SHA256 Credential=AKIDEXAMPLE/20150830/us-east-1/service/aws4_request, '
                . 'SignedHeaders=host;x-amz-date, '
                . 'Signature=5fa00fa31553b73ebf1942676e86291e8372ff2a2260956d9b8aae1d763fbf31',
```

```php
<?php

namespace AetherUpload\Storage;

/**
 * AWS Signature Version 4（S3 规则：路径逐段编码、不归一化点段、不双重编码）。
 * 纯函数；正确性锚 = AWS 官方 sig-v4 测试向量（tests/SignatureV4Test.php）。
 */
final class SignatureV4
{
    private const ALGORITHM = 'AWS4-HMAC-SHA256';

    private function __construct()
    {
    }

    /**
     * @param array $headers 名称大小写不敏感；值 trim + 连续空白折叠；参与签名前按小写名排序
     */
    public static function canonicalRequest(string $method, string $uri, string $query, array $headers, string $payloadHash): string
    {
        $normalized = self::normalizedHeaders($headers);

        $canonicalHeaders = '';
        foreach ( $normalized as $name => $value ) {
            $canonicalHeaders .= $name . ':' . $value . "\n";
        }

        return strtoupper($method) . "\n"
            . self::canonicalUri($uri) . "\n"
            . self::canonicalQuery($query) . "\n"
            . $canonicalHeaders . "\n"
            . implode(';', array_keys($normalized)) . "\n"
            . $payloadHash;
    }

    public static function stringToSign(string $amzDate, string $date, string $region, string $service, string $canonicalRequest): string
    {
        return self::ALGORITHM . "\n"
            . $amzDate . "\n"
            . self::scope($date, $region, $service) . "\n"
            . hash('sha256', $canonicalRequest);
    }

    /** @return string 原始二进制签名密钥 */
    public static function signingKey(string $secretKey, string $date, string $region, string $service): string
    {
        $key = hash_hmac('sha256', $date, 'AWS4' . $secretKey, true);
        $key = hash_hmac('sha256', $region, $key, true);
        $key = hash_hmac('sha256', $service, $key, true);

        return hash_hmac('sha256', 'aws4_request', $key, true);
    }

    public static function authorizationHeader(string $method, string $uri, string $query, array $headers, string $payloadHash, string $accessKey, string $secretKey, string $region, string $service, string $amzDate): string
    {
        $date = substr($amzDate, 0, 8);
        $signature = hash_hmac(
            'sha256',
            self::stringToSign($amzDate, $date, $region, $service, self::canonicalRequest($method, $uri, $query, $headers, $payloadHash)),
            self::signingKey($secretKey, $date, $region, $service)
        );

        return self::ALGORITHM
            . ' Credential=' . $accessKey . '/' . self::scope($date, $region, $service)
            . ', SignedHeaders=' . implode(';', array_keys(self::normalizedHeaders($headers)))
            . ', Signature=' . $signature;
    }

    /** @return string 完整查询串（含 X-Amz-Signature），供 URL 拼接 */
    public static function presignQuery(string $method, string $uri, string $query, array $headers, string $accessKey, string $secretKey, string $region, string $service, string $amzDate, int $expires, string $payloadHash = 'UNSIGNED-PAYLOAD'): string
    {
        $date = substr($amzDate, 0, 8);

        $params = [
            'X-Amz-Algorithm' => self::ALGORITHM,
            'X-Amz-Credential' => $accessKey . '/' . self::scope($date, $region, $service),
            'X-Amz-Date' => $amzDate,
            'X-Amz-Expires' => (string)$expires,
            'X-Amz-SignedHeaders' => implode(';', array_keys(self::normalizedHeaders($headers))),
        ];

        // 把已有查询参数与 X-Amz-* 合并、统一编码一次、按名称（再按值）排序
        $pairs = self::parseQuery($query);
        foreach ( $params as $name => $value ) {
            $pairs[] = [rawurlencode((string)$name), rawurlencode((string)$value)];
        }
        usort($pairs, static function (array $a, array $b): int {
            return $a[0] === $b[0] ? strcmp($a[1], $b[1]) : strcmp($a[0], $b[0]);
        });
        $canonicalQuery = implode('&', array_map(static function (array $p): string {
            return $p[0] . '=' . $p[1];
        }, $pairs));

        $signature = hash_hmac(
            'sha256',
            self::stringToSign($amzDate, $date, $region, $service, self::canonicalRequest($method, $uri, $canonicalQuery, $headers, $payloadHash)),
            self::signingKey($secretKey, $date, $region, $service)
        );

        return $canonicalQuery . '&X-Amz-Signature=' . $signature;
    }

    private static function scope(string $date, string $region, string $service): string
    {
        return $date . '/' . $region . '/' . $service . '/aws4_request';
    }

    /** @return array<string,string> 小写名 => 归一值，按名排序 */
    private static function normalizedHeaders(array $headers): array
    {
        $normalized = [];

        foreach ( $headers as $name => $value ) {
            $normalized[strtolower((string)$name)] = trim((string)preg_replace('/\s+/', ' ', (string)$value));
        }

        ksort($normalized, SORT_STRING);

        return $normalized;
    }

    /** S3：逐段 rawurlencode（不做点段归一化、不双重编码） */
    private static function canonicalUri(string $uri): string
    {
        return implode('/', array_map('rawurlencode', explode('/', $uri)));
    }

    private static function canonicalQuery(string $query): string
    {
        $pairs = self::parseQuery($query);
        usort($pairs, static function (array $a, array $b): int {
            return $a[0] === $b[0] ? strcmp($a[1], $b[1]) : strcmp($a[0], $b[0]);
        });

        return implode('&', array_map(static function (array $p): string {
            return $p[0] . '=' . $p[1];
        }, $pairs));
    }

    /** @return array<int,array{0:string,1:string}> 已编码的 [名, 值] 对（先解码再编码一次，保证规范语义） */
    private static function parseQuery(string $query): array
    {
        if ( $query === '' ) {
            return [];
        }

        $pairs = [];
        foreach ( explode('&', $query) as $pair ) {
            if ( $pair === '' ) {
                continue;
            }
            $eq = strpos($pair, '=');
            $name = $eq === false ? $pair : substr($pair, 0, $eq);
            $value = $eq === false ? '' : substr($pair, $eq + 1);
            $pairs[] = [rawurlencode(rawurldecode($name)), rawurlencode(rawurldecode($value))];
        }

        return $pairs;
    }
}
```

```php
<?php

namespace AetherUpload\Storage;

/**
 * 最小 S3 客户端：SigV4 标准签名 + curl 传输 + 重试 + multipart。
 * 只做本包需要的五个操作：PUT / multipart / HEAD / DELETE / 预签名 GET。
 *
 * 关键设计：成品文件在本地 → payload sha256 可预算 → 永远不用 aws-chunked 流式签名
 * （各家 S3 兼容实现最易翻车处）。$transport 可注入：单测逐字节断言请求，生产用 curlTransport()。
 */
final class S3Client
{
    /** 固定分片大小：非末片恒为 64MB（≥ S3 的 5MB 下限），末片可小 */
    private const PART_SIZE = 67108864;

    private const MAX_ATTEMPTS = 3;
    private const RETRY_SLEEP_US = 200000;

    /** @var array */
    private $config;

    /** @var callable */
    private $transport;

    public function __construct(array $config, ?callable $transport = null)
    {
        $config = array_merge([
            'endpoint' => '',
            'region' => '',
            'bucket' => '',
            'access_key' => '',
            'secret_key' => '',
            'path_style' => true,
            'prefix' => '',
            'multipart_threshold' => 104857600,
        ], $config);

        foreach ( ['region', 'bucket', 'access_key', 'secret_key'] as $required ) {
            if ( ! is_string($config[$required]) || $config[$required] === '' ) {
                // 只报键名，绝不回显值
                throw new \Exception('S3 配置缺少 storage.s3.' . $required . '（S3 驱动必需）');
            }
        }

        $this->config = $config;
        $this->transport = $transport ?: self::curlTransport();
    }

    public function key(string $groupDir, string $groupSubDir, string $name): string
    {
        $prefix = (string)$this->config['prefix'];

        if ( $prefix !== '' && substr($prefix, -1) !== '/' ) {
            $prefix .= '/';
        }

        return $prefix . $groupDir . '/' . $groupSubDir . '/' . $name;
    }

    public function put(string $key, string $localPath, string $contentType): void
    {
        $size = (int)filesize($localPath);
        $response = $this->call('PUT', $key, '', ['Content-Type' => $contentType], null, $localPath, $size, 0);

        if ( $response['status'] !== 200 ) {
            throw $this->fail('PUT', $key, $response);
        }
    }

    public function head(string $key): bool
    {
        $response = $this->call('HEAD', $key, '', [], null, null, null, 0);

        if ( $response['status'] === 200 ) {
            return true;
        }

        if ( $response['status'] === 404 ) {
            return false;
        }

        throw $this->fail('HEAD', $key, $response);
    }

    public function delete(string $key): void
    {
        $response = $this->call('DELETE', $key, '', [], null, null, null, 0);

        // S3 删除幂等：缺失也是成功（204；个别实现 404，同样放过）
        if ( ! in_array($response['status'], [200, 204, 404], true) ) {
            throw $this->fail('DELETE', $key, $response);
        }
    }

    public function presign(string $key, int $ttl, array $responseParams = []): string
    {
        [$scheme, $host, $path] = $this->objectUrl($key);

        $query = '';
        foreach ( $responseParams as $name => $value ) {
            $query .= ($query === '' ? '' : '&') . rawurlencode((string)$name) . '=' . rawurlencode((string)$value);
        }

        $signed = SignatureV4::presignQuery(
            'GET',
            $path,
            $query,
            ['host' => $host],
            $this->config['access_key'],
            $this->config['secret_key'],
            $this->config['region'],
            's3',
            gmdate('Ymd\THis\Z'),
            $ttl
        );

        return $scheme . '://' . $host . $path . '?' . $signed;
    }

    public function uploadMultipartOrPut(string $key, string $localPath, string $contentType): void
    {
        if ( (int)filesize($localPath) > (int)$this->config['multipart_threshold'] ) {
            $this->multipart($key, $localPath, $contentType);

            return;
        }

        $this->put($key, $localPath, $contentType);
    }

    public function multipart(string $key, string $localPath, string $contentType): void
    {
        $init = $this->call('POST', $key, 'uploads=', ['Content-Type' => $contentType], null, null, null, 0);

        if ( $init['status'] !== 200 || ! preg_match('#<UploadId>([^<]+)</UploadId>#', (string)$init['body'], $m) ) {
            throw $this->fail('CreateMultipartUpload', $key, $init);
        }

        $uploadId = html_entity_decode($m[1], ENT_QUOTES | ENT_XML1);
        $fileSize = (int)filesize($localPath);
        $partCount = (int)ceil($fileSize / self::PART_SIZE);
        $parts = [];

        for ( $i = 1; $i <= $partCount; $i++ ) {
            $offset = ($i - 1) * self::PART_SIZE;
            $size = min(self::PART_SIZE, $fileSize - $offset);

            $response = $this->call('PUT', $key, 'partNumber=' . $i . '&uploadId=' . rawurlencode($uploadId), ['Content-Type' => $contentType], null, $localPath, $size, $offset);

            if ( $response['status'] !== 200 || ! isset($response['headers']['etag']) ) {
                $this->abort($key, $uploadId);
                throw $this->fail('UploadPart#' . $i, $key, $response);
            }

            $parts[] = ['number' => $i, 'etag' => $response['headers']['etag']];
        }

        $xml = '<CompleteMultipartUpload>';
        foreach ( $parts as $part ) {
            // ENT_NOQUOTES：ETag 的双引号是值的一部分，不做实体化（生效于 XML 文本节点，&<> 仍被转义）
            $xml .= '<Part><PartNumber>' . $part['number'] . '</PartNumber><ETag>' . htmlspecialchars($part['etag'], ENT_NOQUOTES) . '</ETag></Part>';
        }
        $xml .= '</CompleteMultipartUpload>';

        $complete = $this->call('POST', $key, 'uploadId=' . rawurlencode($uploadId), ['Content-Type' => 'application/xml'], $xml, null, null, 0);

        // 经典陷阱：HTTP 200 但体内嵌 <Error> 仍是失败
        if ( $complete['status'] === 200 && stripos((string)$complete['body'], '<Error') === false ) {
            return;
        }

        // 响应丢失场景：complete 实际已成功但响应没回来 —— HEAD 确认后视为成功
        if ( $this->head($key) ) {
            return;
        }

        $this->abort($key, $uploadId);
        throw $this->fail('CompleteMultipartUpload', $key, $complete);
    }

    /** @return callable 默认 curl 传输；bodyFile 时以 CURLOPT_UPLOAD 流式发送（不载入内存） */
    public static function curlTransport(): callable
    {
        return static function (array $request): array {
            $headers = [];
            foreach ( $request['headers'] as $name => $value ) {
                $headers[] = $name . ': ' . $value;
            }

            $responseHeaders = [];
            $ch = curl_init($request['url']);

            curl_setopt_array($ch, [
                CURLOPT_CUSTOMREQUEST => $request['method'],
                CURLOPT_RETURNTRANSFER => true,
                CURLOPT_HTTPHEADER => $headers,
                CURLOPT_CONNECTTIMEOUT => 10,
                // 无总超时（大文件 PUT 不能被截断），低速守卫兜底
                CURLOPT_LOW_SPEED_LIMIT => 1024,
                CURLOPT_LOW_SPEED_TIME => 60,
                CURLOPT_HEADERFUNCTION => static function ($handle, string $line) use (&$responseHeaders): int {
                    $responseHeaders[] = rtrim($line, "\r\n");

                    return strlen($line);
                },
            ]);

            $handle = null;

            if ( $request['bodyFile'] !== null ) {
                $handle = fopen($request['bodyFile'], 'rb');

                if ( $handle === false ) {
                    throw new \Exception('无法打开本地文件：' . $request['bodyFile']);
                }

                if ( $request['bodyOffset'] > 0 ) {
                    fseek($handle, $request['bodyOffset']);
                }

                // CURLOPT_UPLOAD 隐含 PUT；INFILESIZE 限定只读 size 字节（multipart 分片零拷贝）
                curl_setopt($ch, CURLOPT_UPLOAD, true);
                curl_setopt($ch, CURLOPT_INFILE, $handle);
                curl_setopt($ch, CURLOPT_INFILESIZE, $request['bodySize']);
            } elseif ( $request['body'] !== null ) {
                curl_setopt($ch, CURLOPT_POSTFIELDS, $request['body']);
            }

            if ( $request['method'] === 'HEAD' ) {
                curl_setopt($ch, CURLOPT_NOBODY, true);
            }

            $body = curl_exec($ch);
            $status = (int)curl_getinfo($ch, CURLINFO_RESPONSE_CODE);
            $error = curl_error($ch);

            if ( $handle !== null ) {
                fclose($handle);
            }
            curl_close($ch);

            if ( $body === false ) {
                throw new \Exception('curl: ' . $error);
            }

            $parsed = [];
            foreach ( $responseHeaders as $line ) {
                $pos = strpos($line, ':');
                if ( $pos !== false ) {
                    $parsed[strtolower(substr($line, 0, $pos))] = trim(substr($line, $pos + 1));
                }
            }

            return ['status' => $status, 'headers' => $parsed, 'body' => (string)$body];
        };
    }

    /** @return array{0:string,1:string} [scheme, host, path]（path 以 / 开头，含 bucket 视 path_style） */
    private function objectUrl(string $key): array
    {
        $endpoint = (string)$this->config['endpoint'];
        $pathStyle = $this->config['path_style'] !== false;
        $bucket = (string)$this->config['bucket'];

        if ( $endpoint === '' ) {
            $region = $this->config['region'];
            $host = $pathStyle ? 's3.' . $region . '.amazonaws.com' : $bucket . '.s3.' . $region . '.amazonaws.com';
            $path = $pathStyle ? '/' . $bucket . '/' . $key : '/' . $key;

            return ['https', $host, $path];
        }

        $parts = parse_url(strpos($endpoint, '://') === false ? 'https://' . $endpoint : $endpoint);
        $scheme = $parts['scheme'] ?? 'https';
        $base = rtrim($parts['path'] ?? '', '/');
        $port = isset($parts['port']) ? ':' . $parts['port'] : '';

        if ( $pathStyle ) {
            return [$scheme, ($parts['host'] ?? '') . $port, $base . '/' . $bucket . '/' . $key];
        }

        return [$scheme, $bucket . '.' . ($parts['host'] ?? '') . $port, $base . '/' . $key];
    }

    /**
     * 发一个已签名请求（幂等操作内部重试至多 3 次：网络异常 / 5xx / 429；4xx 不重试）。
     */
    private function call(string $method, string $key, string $query, array $headers, ?string $body, ?string $bodyFile, ?int $bodySize, int $bodyOffset): array
    {
        [$scheme, $host, $path] = $this->objectUrl($key);

        $payloadHash = $bodyFile !== null ? (string)hash_file('sha256', $bodyFile) : hash('sha256', (string)$body);
        // 只签三个自产头：host / x-amz-content-sha256 / x-amz-date（Content-Type 等不签，S3 允许）
        $signedHeaders = [
            'host' => $host,
            'x-amz-content-sha256' => $payloadHash,
            'x-amz-date' => $amzDate = gmdate('Ymd\THis\Z'),
        ];

        $request = [
            'method' => $method,
            'url' => $scheme . '://' . $host . $path . ($query === '' ? '' : '?' . $query),
            'headers' => array_merge(
                $headers,
                [
                    'Host' => $host,
                    'x-amz-content-sha256' => $payloadHash,
                    'x-amz-date' => $amzDate,
                    'Authorization' => SignatureV4::authorizationHeader($method, $path, $query, $signedHeaders, $payloadHash, $this->config['access_key'], $this->config['secret_key'], $this->config['region'], 's3', $amzDate),
                ]
            ),
            'body' => $body,
            'bodyFile' => $bodyFile,
            'bodyOffset' => $bodyOffset,
            'bodySize' => $bodySize,
        ];

        $attempt = 0;

        while ( true ) {
            $attempt++;

            try {
                $response = ($this->transport)($request);
            } catch ( \Exception $e ) {
                if ( $attempt >= self::MAX_ATTEMPTS ) {
                    throw new \Exception('S3 ' . $method . ' failed: transport error: ' . $e->getMessage() . ' (key=' . $key . ')', 0, $e);
                }

                usleep(self::RETRY_SLEEP_US * $attempt);
                continue;
            }

            if ( ($response['status'] >= 500 || $response['status'] === 429) && $attempt < self::MAX_ATTEMPTS ) {
                usleep(self::RETRY_SLEEP_US * $attempt);
                continue;
            }

            return $response;
        }
    }

    private function abort(string $key, string $uploadId): void
    {
        try {
            $this->call('DELETE', $key, 'uploadId=' . rawurlencode($uploadId), [], null, null, null, 0);
        } catch ( \Exception $e ) {
            // best-effort：清理失败不掩盖原始异常
        }
    }

    /** @param array{status:int,body:string} $response */
    private function fail(string $op, string $key, array $response): \Exception
    {
        $body = (string)$response['body'];
        $code = preg_match('#<Code>([^<]*)</Code>#', $body, $m) ? $m[1] : '';
        $message = preg_match('#<Message>([^<]*)</Message>#', $body, $m) ? $m[1] : substr($body, 0, 200);

        return new \Exception(sprintf('S3 %s failed: HTTP %d %s %s (key=%s)', $op, $response['status'], $code, $message, $key));
    }
}
```

```php
<?php

namespace AetherUpload\Storage;

use AetherUpload\Runtime;

/**
 * S3 兼容驱动。对象键 = {prefix}{group_dir}/{subdir}/{name}（与本地布局同构，root_dir 不泄漏进桶）。
 */
final class S3Storage implements StorageInterface
{
    /** 预签名 URL 有效期：下载是「302 后立即取」，300s 足够；不作为配置项 */
    const PRESIGN_TTL = 300;

    /** @var S3Client */
    private $client;

    public function __construct(array $config, ?callable $transport = null)
    {
        $this->client = new S3Client($config, $transport);
    }

    public function publish(string $localPath, string $groupDir, string $groupSubDir, string $name): void
    {
        $key = $this->client->key($groupDir, $groupSubDir, $name);

        try {
            if ( ! $this->client->head($key) ) {
                $contentType = @mime_content_type($localPath);
                $this->client->uploadMultipartOrPut(
                    $key,
                    $localPath,
                    is_string($contentType) && $contentType !== '' ? $contentType : 'application/octet-stream'
                );
            }
        } catch ( \Exception $e ) {
            // 对外统一为上传链路的既有错误键；底层原因挂在 previous 上供排查（消息不含凭据）
            throw new \Exception(Runtime::trans('rename_resource_fail'), 0, $e);
        }

        if ( @unlink($localPath) === false ) {
            throw new \Exception(Runtime::trans('delete_resource_fail'));
        }
    }

    public function exists(string $groupDir, string $groupSubDir, string $name): bool
    {
        return $this->client->head($this->client->key($groupDir, $groupSubDir, $name));
    }

    public function delete(string $groupDir, string $groupSubDir, string $name): bool
    {
        $this->client->delete($this->client->key($groupDir, $groupSubDir, $name));

        return true;
    }

    public function url(string $groupDir, string $groupSubDir, string $name, array $responseParams = []): ?string
    {
        return $this->client->presign($this->client->key($groupDir, $groupSubDir, $name), self::PRESIGN_TTL, $responseParams);
    }
}
```
